onInputChange(e) {

    })

```js
// cloudfunctions/room/index.js
async function rentRoom(params, openId) {
    // 1. 解构参数 - 对应前端传递的字段
    const {
        roomId,
        tenantName,
        tenantPhone,
        lastWaterReading,          // 🔑 对应前端的表读数字段
        lastElectricityReading,
        // ... 其他字段
    } = params
    // 2. 构造更新数据
    const updateData = {
        status: 2,
        tenantName,
        tenantPhone,
        lastWaterReading,          // 🔑 更新到数据库
        lastElectricityReading,
        // ...
    }

    // 3. 更新数据库
    await db.collection('rooms').doc(roomId).update({
        data: updateData
    })

    // 4. 返回结果
    return {
        code: 200,
        message: '房间出租成功'
    }
}
```
